# phix — align a small read set to a reference
#
# A four-step pipeline over the phiX174 genome, the usual sequencing control:
# fetch it, simulate reads from it, align them back, and count what mapped.
#
# Run `just` with no arguments to see every recipe.

set dotenv-load := false

data    := "data"
results := "results"

# Show every recipe in this justfile
default:
    @just --list --unsorted

# Download the phiX174 reference from NCBI
[group('1. data')]
fetch-reference:
    mkdir -p {{data}}
    curl -L -o {{data}}/phix.fa \
      "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/efetch.fcgi?db=nuccore&id=NC_001422.1&rettype=fasta&retmode=text"
    echo "reference: $(grep -c '^>' {{data}}/phix.fa) sequence, $(grep -v '^>' {{data}}/phix.fa | tr -d '\n' | wc -c) bp"

# Simulate a read set from the reference
[group('1. data')]
simulate-reads n="4000" length="150":
    scripts/simulate.py {{data}}/phix.fa {{n}} {{length}} > {{data}}/reads.fq

# Check the tools this pipeline shells out to are present
[private]
_check-tools:
    @for t in curl minibwa samtools; do command -v $t >/dev/null || echo "missing $t"; done

alias reads := simulate-reads

mod align
